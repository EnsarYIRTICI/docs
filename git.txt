init -->
	git init
	git remote add origin https://github.com/EnsarYIRTICI/<repository-name>.git
	git add .
	git commit -m "first"
	git push origin master


update -->
	git add .
	git commit -m "bugfix"
	git push origin master


pull changes --> git pull


remote -->
	ls --> git remote -v
	add --> git remote add origin https://github.com/EnsarYIRTICI/<repository-name>.git
	change --> git remote set-url origin https://github.com/EnsarYIRTICI/<repository-name>.git
	del --> git remote rm origin


stop follow --> git rm -r --cached .


config -->
	small/large case --> git config core.ignorecase false

branch -->
	ls --> git branch

github -->
	 login -->
		sudo apt update
		sudo apt install gh
		gh auth login


feat: New feature
fix: Bug fix

refactor: Code cleanup without changing behavior (e.g. remove unused function)
chore: Maintenance tasks (config, dependency updates, project setup)
docs: Documentation changes
style: Code style changes (formatting, spacing, punctuation, linter compliance)
test: Add or update tests
perf: Performance improvements

build: Build system or dependency-related changes (pubspec.yaml, gradle, package.json, etc.)
ci: CI/CD configuration or script changes (GitHub Actions, GitLab CI, Jenkins)

revert: Revert a previous commit
release: Version bump, changelog updates (some teams use chore(release))

merge: Branch merges (usually auto-generated)
deps: Dependency updates (more specific alternative to chore)
hotfix: Urgent production fix (some teams combine this with fix)
security: Security vulnerability fixes
design / ui: UI/UX changes (colors, icons, layout)
